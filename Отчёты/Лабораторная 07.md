## Основные сведения

k=⌊√p⌋+1. Строим yᵢ=a^(ik) mod p и zⱼ=b·aʲ mod p. Если yᵢ=zⱼ, то проверяем x=ik−j подстановкой в aˣ mod p.

```python
from arithmetic import integer_root, power_mod

A, B, P = 2, 12161, 30203


def baby_giant(a, b, p):
    k = integer_root(p)+1
    giant = power_mod(a, k, p)
    y, z = 1, b % p
    baby = {}
    for j in range(k):
        baby.setdefault(z, j)
        z = z*a % p
    for i in range(k+1):
        if y in baby:
            j = baby[y]
            x = i*k-j
            if x >= 0 and power_mod(a, x, p) == b % p:
                return k, i, j, y, x
        y = y*giant % p
    return None


if __name__ == "__main__":
    k, i, j, residue, log = baby_giant(A, B, P)
    print(f"a={A}, b={B}, p={P}, k={k}")
    print(f"i={i}, j={j}, y_i=z_j={residue}, x={log}")
    print(f"Проверка: {A}^{log} mod {P} = {power_mod(A, log, P)}")
```

def bezout(a, b):
    r, s = abs(a), abs(b)
    x, u, y, v = 1, 0, 0, 1
    while s:
        q, rem = divmod(r, s)
        r, s = s, rem
        x, u = u, x - q * u
        y, v = v, y - q * v
    return r, x if a >= 0 else -x, y if b >= 0 else -y


def inverse(a, m):
    if m < 2:
        raise ValueError("Модуль должен быть не меньше 2")
    g, x, _ = bezout(a, m)
    if g != 1:
        raise ValueError(f"Обратного элемента нет: НОД = {g}")
    return x % m


def power_mod(base, exponent, modulus):
    if exponent < 0 or modulus < 1:
        raise ValueError("Нужны неотрицательная степень и положительный модуль")
    value = 1 % modulus
    base %= modulus
    while exponent:
        if exponent & 1:
            value = (value * base) % modulus
        base = base * base % modulus
        exponent >>= 1
    return value


def integer_root(n):
    if n < 0:
        raise ValueError("Корень определён для n ≥ 0")
    if n < 2:
        return n
    x = n
    while (y := (x + n // x) // 2) < x:
        x = y
    return x


ALPHABET = "абвгдежзийклмнопрстуфхцчшщъыьэюя"


def russian_letters(text):
    return ''.join(c for c in text.lower().replace('ё', 'е') if c in ALPHABET)
